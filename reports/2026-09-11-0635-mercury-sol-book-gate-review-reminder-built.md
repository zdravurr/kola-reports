## 1. What was built

### 1a. The conditions — §5b verbatim

| condition | fires | how often |
|---|---|---|
| review point | n_LONG + n_SHORT ≥ 200, **or** the UTC date reaches 2026-10-11, whichever comes first | **once** |
| side wire | a side has n ≥ 50 **and** refused/n above 5 % | **immediately, once per side** |
| ratio wire | both sides have n ≥ 50 **and** the side ratio is above 2× | **immediately, once** |

- **The comparisons use integers.** For example, `r·100 > 5·n` and `r_hi·n_lo > 2·r_lo·n_hi`. So exactly 5 % and exactly 2× do **not** fire ("above", as registered), and there is no float edge.
- **One side at 0 refusals while the other has some** counts as a ratio of ∞, which is a breach. That is how 08-14's SHORT clause B hid at 0/481. Both sides at 0 is not a breach.
- **Every condition fires once, whichever way it becomes due.** The state file's `fired` map records each key with its time and counts. To re-arm a condition, delete its key.
- **A failed send is not marked**, so the next pass retries it 30 min later.

The query, character for character as §5b — the contract holds it against the 0550 report file:

```sql
SELECT CASE WHEN signal_type='open_long' THEN 'LONG' ELSE 'SHORT' END AS side,
       COUNT(*) AS n, SUM(status='book_blocked') AS refused
FROM trades
WHERE (book_gate_lean IS NOT NULL OR status='book_blocked')
  AND timestamp >= '2026-09-11 05:40:08'
GROUP BY side;
```

On the live database, only `open_long` / `open_short` rows ever carry book-gate facts (569 / 565 over all time), so the `ELSE 'SHORT'` branch cannot mislabel another signal type.

def connect_ro(db):
    uri = "file:" + urllib.parse.quote(os.path.abspath(db)) + "?mode=ro"
    return sqlite3.connect(uri, uri=True, timeout=10)


def read_counts(db):
    """{'LONG': (n, refused), 'SHORT': (n, refused)} — the §5b query, read-only."""
    con = connect_ro(db)

def rate_breached(n, r):

def ratio_breached(counts):
    (nl, rl), (ns, rs) = counts["LONG"], counts["SHORT"]
    if nl < WIRE_MIN_N or ns < WIRE_MIN_N:
        return False
    # rate_hi > 2 × rate_lo  ⇔  r_hi·n_lo > 2·r_lo·n_hi   (0 vs >0 is a breach; 0 vs 0 is not)
    return rl * ns > WIRE_RATIO * rs * nl or rs * nl > WIRE_RATIO * rl * ns


def ratio_value(counts):
    """(ratio, higher side) — ratio is None when neither side refused, inf when one side is 0."""
    (nl, rl), (ns, rs) = counts["LONG"], counts["SHORT"]
    a = rl / nl if nl else 0.0
    b = rs / ns if ns else 0.0
    hi, lo, side = (a, b, "LONG") if a >= b else (b, a, "SHORT")
    if hi == 0:

    return (float("inf") if lo == 0 else hi / lo), side

    return [comb(n, k) * p ** k * (1 - p) ** (n - k) for k in range(n + 1)]

    """P(refused/n > 5 %) by noise alone."""
    return sum(pk for k, pk in enumerate(_pmf(n, p)) if k * 100 > WIRE_RATE_PCT * n)

    """P(side ratio > 2×) by noise alone, both sides at their registered rates."""
    A, B = _pmf(nl, EXPECTED["LONG"]), _pmf(ns, EXPECTED["SHORT"])
    t = 0.0
    for a, pa in enumerate(A):
        for b, pb in enumerate(B):
            if a * ns > WIRE_RATIO * b * nl or b * nl > WIRE_RATIO * a * ns:

    if ratio_breached(counts):
        x, side = ratio_value(counts)
        out["wire_ratio"] = (f"WIRE TRIPPED: side ratio {_x(x)} ({side} higher) is above "
                             f"{WIRE_RATIO}× with both sides at n >= {WIRE_MIN_N}")
    return out


def floors_in_config(path):
    """BOOK_GATE_LEAN_FLOOR from SOL config.py, read as TEXT (never imported). None if unreadable."""
    try:
        tree = ast.parse(open(path, encoding="utf-8").read())
    except (OSError, SyntaxError, ValueError):
        return None
    for node in tree.body:
        targets = node.targets if isinstance(node, ast.Assign) else (
            [node.target] if isinstance(node, ast.AnnAssign) else [])
        if any(isinstance(t, ast.Name) and t.id == "BOOK_GATE_LEAN_FLOOR" for t in targets):
            try:
                return ast.literal_eval(node.value)
            except ValueError:
                return None
    return None


# ── the alert text ──────────────────────────────────────────────────────────────────────
def _pct(v):
    return f"{v * 100:.2f} %"


def _x(v):
    return "∞" if v == float("inf") else f"{v:.2f}×"


def format_alert(reasons, counts, now):
    (nl, rl), (ns, rs) = counts["LONG"], counts["SHORT"]
    total = nl + ns
    x, side = ratio_value(counts)
    if x is None:
        ratio_line = "Side ratio: n/a (no refusals on either side)"
    elif x == float("inf"):
        lo = "SHORT" if side == "LONG" else "LONG"
        ratio_line = f"Side ratio: ∞ ({lo} has 0 refusals) · registered {EXPECTED_RATIO}×"
    else:
        ratio_line = f"Side ratio: {_x(x)} ({side} higher) · registered {EXPECTED_RATIO}×"

    def side_line(s, n, r):
        v = _pct(r / n) if n else "n/a"
        return f"{s} {r}/{n} refused = {v} · registered {_pct(EXPECTED[s])}"

    noise = []
    if nl:
        noise.append(f"LONG above 5 % {_pct(p_rate_above(nl, EXPECTED['LONG']))}")
    if ns:
        noise.append(f"SHORT above 5 % {_pct(p_rate_above(ns, EXPECTED['SHORT']))}")
    if nl and ns:
        noise.append(f"ratio above 2× {_pct(p_ratio_above(nl, ns))}")

    lines = [
        "[Mercury-SOL] BOOK-GATE REVIEW — ALERT",
        *reasons,
        f"Counted since the 09-11 re-cut, {ANCHOR} UTC · now {now:%Y-%m-%d %H:%M} UTC · "
        f"{total}/{REVIEW_N} evaluations",
        side_line("LONG", nl, rl),
        side_line("SHORT", ns, rs),
        ratio_line,
        "Wires (pre-registered 09-11, unchanged): above 5 % on a side, or a side ratio above 2× "
        "→ re-cut the ruler per side and say so. Never loosen quietly.",
        "FALSE-ALARM LINE (0550 §5a): if the registered rates are true, the chance of a false "
        "alarm at the first review, about 100 per side, is LONG 11.7 %, SHORT 5.4 %. A LONG "
        "reading between 5 % and 6 % at the first review is within noise. Nobody re-cuts on "
        "that without reading this line.",
        # Sequential looks (Monte C., 40 000 paths, seed 20260911, registered rates true,
        # n equal per side, checked at every evaluation): P(any wire ever trips by noise)
        # = 0.848 by 100/side (the first review), 0.933 by 540/side (≈ 2026-10-11).
        "SEQUENTIAL LOOKS: the wires are armed from n = 50 and checked every 30 min, so noise "
        "alone trips at least one of them before the first review with ≈ 85 % probability, and "
        "by 2026-10-11 with ≈ 93 % (Monte C., registered rates true). A wire alert is a "
        "prompt to look, not a finding.",
    ]
    if noise:
        lines.append(f"Noise at this n, registered rates assumed true: {' · '.join(noise)}.")
    lines.append("Canon: mercury-sol/OPEN-ITEMS-SOL.md §BOOK-GATE-RECUT-2026-09-11. Alert, not a "
                 "report: each condition fires once (state .kola_state/sol_book_gate_review.json).")
    return "\n".join(lines)


def format_stale(floors):
    return "\n".join([
        "[Mercury-SOL] BOOK-GATE REVIEW COUNTER IS STALE — ALERT",
        f"SOL config.py BOOK_GATE_LEAN_FLOOR is now {floors}; the review counter is pinned to the "
        f"09-11 registration {PINNED_FLOORS}, counted from {ANCHOR}.",
        "Its counts no longer measure the running gate. Re-point ANCHOR, EXPECTED and "
        "PINNED_FLOORS in workspace scripts/sol_book_gate_review.py to the new registration, "
        "or the next review is missed the way 08-20 was.",
    ])


def format_broken(err):
    return "\n".join([
        "[Mercury-SOL] BOOK-GATE REVIEW COUNTER CANNOT READ trades.db — ALERT",
        f"The read failed twice, {RECHECK_S} s apart: {err}",
        "While this lasts, the review point and the wires are NOT being watched. "
        "Fires once per outage; it re-arms after the next good read.",
    ])


# ── state ───────────────────────────────────────────────────────────────────────────────
def load_state(path):
    try:
        with open(path, encoding="utf-8") as f:
            st = json.load(f)
    except FileNotFoundError:
        st = {}
    except (OSError, ValueError) as e:
        # A corrupt state must not mean silence: move it aside and start clean. The cost is
        # at most one repeated alert per condition, never a missed one.
        _log(f"state unreadable ({e}); moved aside, starting clean")
        try:
            os.replace(path, path + ".corrupt")
        except OSError:
            pass
        st = {}
    st.setdefault("fired", {})
    return st


def save_state(path, st):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    tmp = path + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        json.dump(st, f, indent=2, sort_keys=True)
    os.replace(tmp, path)


# ── one pass ────────────────────────────────────────────────────────────────────────────
def run(db, state_path, send, now=None, config_path=SOL_CONFIG,
        recheck_s=RECHECK_S, sleep=time.sleep):
    """One evaluation. `send(text) -> bool` is injected: no default, so a test cannot reach
    Telegram by forgetting to pass one. Returns (exit_code, [keys fired this pass])."""
    now = now or datetime.now(timezone.utc)
    st = load_state(state_path)
    fired = st["fired"]
    stamp = now.strftime("%Y-%m-%d %H:%M:%S")

    try:
        counts = read_counts(db)
    except sqlite3.Error as first:
        _log(f"read failed ({first}); re-asking in {recheck_s}s before calling anyone")
        sleep(recheck_s)
        try:
            counts = read_counts(db)
        except sqlite3.Error as e:
            if st.get("reader_broken"):
                _log(f"read still failing ({e}); already alerted this outage")
                return 1, []
            ok = bool(send(format_broken(e)))
            _log(f"read failed twice ({e}); alert DELIVERED={ok}")
            if ok:
                st["reader_broken"] = stamp
                save_state(state_path, st)
            return 1, (["reader_broken"] if ok else [])
    st.pop("reader_broken", None)

    new = {k: v for k, v in due_conditions(counts, now).items() if k not in fired}
    out, code = [], 0
    if new:
        text = format_alert(list(new.values()), counts, now)
        ok = bool(send(text))
        if ok:
            for k, v in new.items():
                fired[k] = {"at": stamp, "reason": v,
                            "counts": {s: list(counts[s]) for s in SIDES}}
            out += list(new)
        else:
            code = 1   # not marked: the next pass retries
        _log(f"due={list(new)} DELIVERED={ok}")

    floors = floors_in_config(config_path)
    if floors is not None and floors != PINNED_FLOORS:
        key = "registration_moved:" + json.dumps(floors, sort_keys=True)
        if key not in fired:
            ok = bool(send(format_stale(floors)))
            if ok:
                fired[key] = {"at": stamp, "floors": floors}
                out.append(key)
            else:
                code = 1
            _log(f"config floors {floors} != pinned {PINNED_FLOORS}; alert DELIVERED={ok}")

    (nl, rl), (ns, rs) = counts["LONG"], counts["SHORT"]
    x, _ = ratio_value(counts)
    st["last_run"] = stamp
    st["last_counts"] = {s: list(counts[s]) for s in SIDES}
    save_state(state_path, st)
    _log(f"n={nl + ns}/{REVIEW_N} LONG {rl}/{nl} SHORT {rs}/{ns} "
         f"ratio={'n/a' if x is None else _x(x)} fired_ever={sorted(fired)} new={out}")
    return code, out


def _send_real(text):
    sys.dont_write_bytecode = True            # nothing written into the Titan tree
    from dotenv import load_dotenv
    load_dotenv(os.path.join(SENDER_DIR, ".env"))
    sys.path.insert(0, SENDER_DIR)
    from full_report import send_full_report
    # require_link=False — this is an ALERT, not a report (link gate 04.08; 0550 §5b).
    return send_full_report(text, require_link=False)


def selftest():
    """Read-only proof of every prerequisite the cron pass needs. Sends nothing, writes nothing."""
    ok = True
    try:
        c = read_counts(DB)
        print(f"db: OK mode=ro · LONG {c['LONG'][1]}/{c['LONG'][0]} · SHORT {c['SHORT'][1]}/{c['SHORT'][0]}"
              f" · due now: {sorted(due_conditions(c, datetime.now(timezone.utc)))}")
    except sqlite3.Error as e:
        print(f"db: FAIL {e}")
        ok = False
    floors = floors_in_config(SOL_CONFIG)
    match = floors == PINNED_FLOORS
    print(f"config floors: {floors} · pinned {PINNED_FLOORS} · match={match}")
    ok = ok and match
    print(f"state: {STATE} · dir writable={os.access(os.path.dirname(STATE), os.W_OK)}")
    try:
        sys.dont_write_bytecode = True
        from dotenv import load_dotenv
        load_dotenv(os.path.join(SENDER_DIR, ".env"))
        sys.path.insert(0, SENDER_DIR)
        import full_report
        full_report._get_config()             # raises if token/chat are missing
        print("sender: OK (full_report imported, token and chat id present; nothing sent)")
    except Exception as e:  # noqa: BLE001 — any failure here means the alert cannot leave
        print(f"sender: FAIL {type(e).__name__}: {e}")
        ok = False
    print("SELFTEST", "OK" if ok else "FAIL")
    return 0 if ok else 1


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--selftest", action="store_true")
    a = ap.parse_args()
    if a.selftest:
        return selftest()
    return run(DB, STATE, send=_send_real)[0]


if __name__ == "__main__":
    sys.exit(main())
